# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def middleNode(self, head: ListNode | None) -> ListNode | None:
        # 核心思想：快慢指针（双指针），通过两个指针不同的移动速度，让慢指针最终停在链表中间。
        # 链表长度为单数：fast停在最后一个节点
        # 链表长度为双数：fast停在最后节点后面的空节点,此时slow正好是第二个中间节点
        # 循环终止条件——fast指向空或者fast.next指向空
        fast = head
        slow =head

        while fast and fast.next:
            fast = fast.next.next
            slow = slow.next

        return slow
