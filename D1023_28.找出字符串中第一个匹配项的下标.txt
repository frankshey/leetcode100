class Solution:
    def strStr(self, haystack: str, needle: str) -> int:
        # 学习KMP算法，精彩讲解视频已经收藏到b占【研一上】收藏夹,next数组的视觉逻辑已经更新到github

        # 构建next数组,pattern指模式串，即需要匹配的子串
        def get_next(pattern):
            n = len(pattern)
            next_arr = [0]*n
            j = 0 # 前缀末尾指针
            for i in range(1,n):
                # 不相等，回退j
                while j > 0 and pattern[i] != pattern[j]:
                    j = next_arr[j - 1]
                if pattern[i]  == pattern[j]:
                    j += 1
                next_arr[i] = j
            
            return next_arr

        if not needle:
            return -1

        next_arr = get_next(needle)
        j = 0  # 模式串的指针

        for i in range(len(haystack)):
            #字符不匹配，j回退
            while j > 0 and haystack[i] != needle[j]:
                j = next_arr[j - 1]
            
            if haystack[i] == needle[j]:
                j += 1
            #当j走到模式串的最后一位的时候匹配成功
            if j == len(needle):
                return i - len(needle) + 1

            #上述都没有触发return则说明匹配失败
        return -1

class Solution:
    def strStr(self, haystack: str, needle: str) -> int:
        
        def get_next(pattern) -> str:
            next_arr = [0] * len(pattern)
            j = 0
            # 字母j同时有两个作用，既是已经匹配的最大的前后缀的长度，也是要即将与末尾新加入元素相匹配的后一位元素的下标，而指针i每次+1，即每次for循环，都相当于一个新元素加入到了字符串pattern的末尾。
            # i从1开始，因为pattern的第一位没有前后缀，肯定是0
            for i in range(1,len(pattern)):

                while j > 0 and pattern[j] != pattern[i]:

                    j = next_arr[j-1]

                if pattern[j] == pattern[i]:
                    j += 1
                
                next_arr[i] = j

            return next_arr

        next_arr = get_next(needle)

        n  = len(haystack)
        j = 0
        # i 用来作为主串的指针，j在这里的作用与上一个函数其实本质类似
        for i in range(0,n):

            while j > 0 and haystack[i] != needle[j]:
                #这一位不相等，那么需要回退比较
                j = next_arr[j-1]

            if haystack[i] == needle[j]:
                #这一位相等，那么该比较下一位了
                j += 1

            if j == len(needle):
                #这时候说明模式串的最后一位已经和主串匹配上了,之所以不用长度减去1是因为第一个if
                index = i - j + 1

                return index
        #如果最终匹配失败，那说明j的值不等于len(needle)，但还可以优化，可以直接return-1，因为上面已经判断过是否相等，如果能走到下面这一步，则说明j与len（needle）必然不相等

        if j != len(needle):
            return -1
