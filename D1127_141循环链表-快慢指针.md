# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution:
    def hasCycle(self, head: Optional[ListNode]) -> bool:
        # 快慢指针，快指针走两步，相对速度是1，所以一定能与慢指针相遇
        # 如果无环，fast走到最后循环直接结束。如果有环，那fast和slow一定能相遇

        slow = head
        fast = head
        while fast and fast.next: 
            #fast.next仍然需要判断，防止如果是空的话导致下面fast报错
            slow = slow.next
            fast = fast.next.next
            if fast is slow:
                return True
        return  False
