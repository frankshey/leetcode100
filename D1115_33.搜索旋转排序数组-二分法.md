class Solution:
    def search(self, nums: list[int], target: int) -> int:
        # 和153很相似，同样数组一定有两段单调的区间，之间存在破坏单调性的断点。153是找到这个最小值，最小值一定是断点。
        # 解法1：第1次二分找到最小值，与其比较后，第2次二分找target

        left = 0 
        right = len(nums) - 1

        while left < right:
            m = (left + right) // 2
            if nums[m] > nums[right]: #存在断点，一定在m右侧
                left = m + 1
            else: # 题目没有重复元素，不用考虑相等。
                # m 要么在断点右侧，要么本身是断点
                right = m
        # 此时nums[right]就是最小值，此时需要结合三者共同判断搜索最小值的左侧还是有右侧，即nums[right]、target、和nums[-1]


        if nums[right] > target:
            return -1
        else:
            l = right
            r = len(nums) - 1

            while l < r:
                m = (l + r) // 2
                if nums[m] > target:
                    r = m - 1
                else:
                    l = m + 1
            return   l - 1
