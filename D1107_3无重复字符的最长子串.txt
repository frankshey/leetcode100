#仍然可以用滑动窗口思想来解决。只要是暴力思想类似的

class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        #这个要用来记录各个字符出现字数，适合用哈希表/字典
        #Counter 是 Python 提供的统计工具，帮你记录“某个元素出现了多少次”。
        count = Counter()
        maxLen = 0
        left = 0

        for right,x in enumerate(s):
            count[x] += 1

            while count[x] > 1:
                count[s[left]] -= 1 #注意这个地方容易落了
                left += 1

            maxLen = max(maxLen, right - left + 1)
        return maxLen
